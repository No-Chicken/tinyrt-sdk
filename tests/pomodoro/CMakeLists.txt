cmake_minimum_required(VERSION 3.20)
project(tinyrt_sdk_pomodoro_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(TINYRT_CORE_DIR "" CACHE PATH "Explicit TinyRT core checkout")
set(POMODORO_WASM "" CACHE FILEPATH "SDK-built production 25m/5m Wasm")
set(POMODORO_FAST_WASM "" CACHE FILEPATH "SDK-built test-only 25s/5s Wasm")
if(NOT EXISTS "${TINYRT_CORE_DIR}/cmake/HostWamr.cmake")
  message(FATAL_ERROR "Pass -DTINYRT_CORE_DIR=<compatible TinyRT core checkout>")
endif()
if(NOT EXISTS "${POMODORO_WASM}" OR NOT EXISTS "${POMODORO_FAST_WASM}")
  message(FATAL_ERROR "Build both app.json and app-fast.json; pass POMODORO_WASM and POMODORO_FAST_WASM")
endif()
include("${TINYRT_CORE_DIR}/cmake/HostWamr.cmake")
add_executable(test_pomodoro test_pomodoro.c)
target_compile_definitions(test_pomodoro PRIVATE TEST_WAMR=1)
target_link_libraries(test_pomodoro PRIVATE tinyrt_runtime)
if(MSVC)
  target_compile_options(test_pomodoro PRIVATE /utf-8 /W4 /WX /wd4996)
endif()
enable_testing()
add_test(NAME pomodoro_wamr COMMAND test_pomodoro "${POMODORO_WASM}")
add_test(NAME pomodoro_fast_wamr COMMAND test_pomodoro "${POMODORO_FAST_WASM}" --fast)
set_tests_properties(pomodoro_wamr pomodoro_fast_wamr PROPERTIES TIMEOUT 30)
