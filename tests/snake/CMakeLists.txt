cmake_minimum_required(VERSION 3.20)
project(tinyrt_sdk_snake_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(TINYRT_CORE_DIR "" CACHE PATH "Explicit TinyRT core checkout")
set(SNAKE_WASM "" CACHE FILEPATH "SDK-built Snake Wasm")
if(NOT EXISTS "${TINYRT_CORE_DIR}/cmake/HostWamr.cmake")
  message(FATAL_ERROR "Pass -DTINYRT_CORE_DIR=<compatible TinyRT core checkout>")
endif()
if(NOT EXISTS "${SNAKE_WASM}")
  message(FATAL_ERROR "Build examples/snake and pass -DSNAKE_WASM=<app.wasm>")
endif()
include("${TINYRT_CORE_DIR}/cmake/HostWamr.cmake")
add_executable(test_snake test_snake.c)
target_compile_definitions(test_snake PRIVATE TEST_WAMR=1)
target_link_libraries(test_snake PRIVATE tinyrt_runtime)
if(MSVC)
  target_compile_options(test_snake PRIVATE /utf-8 /W4 /WX /wd4996)
endif()
enable_testing()
add_test(NAME snake_wamr COMMAND test_snake "${SNAKE_WASM}")
set_tests_properties(snake_wamr PROPERTIES TIMEOUT 60)
