cmake_minimum_required(VERSION 3.16)
project(nes_maze_wamr_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
if(NOT TINYRT_ROOT OR NOT WAMR_ROOT_DIR)
  message(FATAL_ERROR "Set TINYRT_ROOT and WAMR_ROOT_DIR explicitly")
endif()
include("${TINYRT_ROOT}/cmake/HostWamr.cmake")
add_executable(test_nes_maze_wamr test_wamr.c)
target_link_libraries(test_nes_maze_wamr PRIVATE tinyrt_runtime)
if(MSVC)
 target_compile_options(test_nes_maze_wamr PRIVATE /utf-8 /W4 /WX /wd4996)
endif()
enable_testing()
if(NES_MAZE_WASM)
 add_test(NAME nes_maze COMMAND test_nes_maze_wamr "${NES_MAZE_WASM}" 100000)
 set_tests_properties(nes_maze PROPERTIES TIMEOUT 120)
endif()
